      173 +def diagnose_column_correlation(
      174 +    params,
      175 +    structure,
      176 +    batch: Dict[str, jnp.ndarray],
      177 +    rng_key: jax.Array,
      178 +) -> Dict[str, float]:
      179 +    """
      180 +    Compute the maximum off-diagonal pairwise Pearson correlation between
      181 +    column z_latent vectors after inference, on a fixed diagnostic batch.
      182 +
      183 +    High correlation across columns means the four (or N) columns are producing
      184 +    redundant features. A sum combiner with redundant column outputs collapses
      185 +    to a single feature; that is the load-bearing claim of Path D in
      186 +    docs/dev-plans/2026-06-21-claude-columnar-cifar10-path-forward.md.
      187 +
      188 +    Returns summary: max off-diagonal |corr|, mean off-diagonal |corr|.
      189 +    """
      190 +    _, _, final_state = get_graph_param_gradient(params, batch, structure, rng_key)
      191 +    col_names = sorted(n for n in final_state.nodes if n.startswith("col_"))
      192 +    if len(col_names) < 2:
      193 +        return {"max_abs_offdiag": 0.0, "mean_abs_offdiag": 0.0}
      194 +
      195 +    flat_cols = []
      196 +    for name in col_names:
      197 +        z = final_state.nodes[name].z_latent
      198 +        # collapse over batch+spatial axes; one row per column
      199 +        flat = jnp.reshape(z, (-1,))
      200 +        flat = flat - jnp.mean(flat)
      201 +        norm = jnp.linalg.norm(flat) + 1e-8
      202 +        flat_cols.append(flat / norm)
      203 +
      204 +    mat = jnp.stack(flat_cols, axis=0)  # (n_cols, n_features)
      205 +    corr = jnp.matmul(mat, mat.T)
      206 +    n = corr.shape[0]
      207 +    mask = 1.0 - jnp.eye(n)
      208 +    off = jnp.abs(corr) * mask
      209 +    max_off = float(jnp.max(off))
      210 +    mean_off = float(jnp.sum(off) / jnp.sum(mask))
      211 +    return {"max_abs_offdiag": max_off, "mean_abs_offdiag": mean_off}

      173 +def diagnose_column_correlation(
      174 +    params,
      175 +    structure,
      176 +    batch: Dict[str, jnp.ndarray],
      177 +    rng_key: jax.Array,
      178 +) -> Dict[str, float]:
      179 +    """
      180 +    Compute the maximum off-diagonal pairwise Pearson correlation between
      181 +    column z_latent vectors after inference, on a fixed diagnostic batch.
      182 +
      183 +    High correlation across columns means the four (or N) columns are producing
      184 +    redundant features. A sum combiner with redundant column outputs collapses
      185 +    to a single feature; that is the load-bearing claim of Path D in
      186 +    docs/dev-plans/2026-06-21-claude-columnar-cifar10-path-forward.md.
      187 +
      188 +    Returns summary: max off-diagonal |corr|, mean off-diagonal |corr|.
      189 +    """
      190 +    _, _, final_state = get_graph_param_gradient(params, batch, structure, rng_key)
      191 +    col_names = sorted(n for n in final_state.nodes if n.startswith("col_"))
      192 +    if len(col_names) < 2:
      193 +        return {"max_abs_offdiag": 0.0, "mean_abs_offdiag": 0.0}
      194 +
      195 +    flat_cols = []
      196 +    for name in col_names:
      197 +        z = final_state.nodes[name].z_latent
      198 +        # collapse over batch+spatial axes; one row per column
      199 +        flat = jnp.reshape(z, (-1,))
      200 +        flat = flat - jnp.mean(flat)
      201 +        norm = jnp.linalg.norm(flat) + 1e-8
      202 +        flat_cols.append(flat / norm)
      203 +
      204 +    mat = jnp.stack(flat_cols, axis=0)  # (n_cols, n_features)
      205 +    corr = jnp.matmul(mat, mat.T)
      206 +    n = corr.shape[0]
      207 +    mask = 1.0 - jnp.eye(n)
      208 +    off = jnp.abs(corr) * mask
      209 +    max_off = float(jnp.max(off))
      210 +    mean_off = float(jnp.sum(off) / jnp.sum(mask))
      211 +    return {"max_abs_offdiag": max_off, "mean_abs_offdiag": mean_off}

      173 +def diagnose_column_correlation(
      174 +    params,
      175 +    structure,
      176 +    batch: Dict[str, jnp.ndarray],
      177 +    rng_key: jax.Array,
      178 +) -> Dict[str, float]:
      179 +    """
      180 +    Compute the maximum off-diagonal pairwise Pearson correlation between
      181 +    column z_latent vectors after inference, on a fixed diagnostic batch.
      182 +
      183 +    High correlation across columns means the four (or N) columns are producing
      184 +    redundant features. A sum combiner with redundant column outputs collapses
      185 +    to a single feature; that is the load-bearing claim of Path D in
      186 +    docs/dev-plans/2026-06-21-claude-columnar-cifar10-path-forward.md.
      187 +
      188 +    Returns summary: max off-diagonal |corr|, mean off-diagonal |corr|.
      189 +    """
      190 +    _, _, final_state = get_graph_param_gradient(params, batch, structure, rng_key)
      191 +    col_names = sorted(n for n in final_state.nodes if n.startswith("col_"))
      192 +    if len(col_names) < 2:
      193 +        return {"max_abs_offdiag": 0.0, "mean_abs_offdiag": 0.0}
      194 +
      195 +    flat_cols = []
      196 +    for name in col_names:
      197 +        z = final_state.nodes[name].z_latent
      198 +        # collapse over batch+spatial axes; one row per column
      199 +        flat = jnp.reshape(z, (-1,))
      200 +        flat = flat - jnp.mean(flat)
      201 +        norm = jnp.linalg.norm(flat) + 1e-8
      202 +        flat_cols.append(flat / norm)
      203 +
      204 +    mat = jnp.stack(flat_cols, axis=0)  # (n_cols, n_features)
      205 +    corr = jnp.matmul(mat, mat.T)
      206 +    n = corr.shape[0]
      207 +    mask = 1.0 - jnp.eye(n)
      208 +    off = jnp.abs(corr) * mask
      209 +    max_off = float(jnp.max(off))
      210 +    mean_off = float(jnp.sum(off) / jnp.sum(mask))
      211 +    return {"max_abs_offdiag": max_off, "mean_abs_offdiag": mean_off}
